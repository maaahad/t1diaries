3. crates/error — shared error type

Read first: Rust Book Ch.9 — Error Handling (all three sub-sections: unrecoverable errors with panic!, recoverable errors with Result, and "to panic or not to panic"). Also skim Ch.6 — Enums and Pattern Matching since AppError is an enum.

crates/error/src/lib.rs 
use async_graphql::{Error as GqlError, ErrorExtensions};
          use thiserror::Error;

          #[derive(Debug, Error)]
          pub enum AppError {
          #[error("not found")]
          NotFound,

          #[error("validation failed: {0}")]
          Validation(String),

          #[error("unauthenticated")]
          Unauthenticated,

          #[error("forbidden")]
          Forbidden,

          #[error("conflict: {0}")]
          Conflict(String),

          #[error("internal error")]
          Internal(#[from] anyhow::Error),

          #[error("database error")]
          Database(#[from] sqlx::Error),
          }

          // Single place that maps our domain errors into GraphQL errors with an
          // `extensions.code` the frontend can branch on (e.g. "UNAUTHENTICATED").
          impl ErrorExtensions for AppError {
          fn extend(&self) -> GqlError {
          GqlError::new(self.to_string()).extend_with(|_, e| {
          let code = match self {
          AppError::NotFound => "NOT_FOUND",
          AppError::Validation(_) => "VALIDATION",
          AppError::Unauthenticated => "UNAUTHENTICATED",
          AppError::Forbidden => "FORBIDDEN",
          AppError::Conflict(_) => "CONFLICT",
          AppError::Database(_) | AppError::Internal(_) => "INTERNAL",
          };
          e.set("code", code);
          })
          }
          }

          pub type AppResult<T> = Result<T, AppError>;

Note sqlx = { workspace = true } also needs adding to this crate's dependencies for the #[from] sqlx::Error conversion to compile — add it alongside anyhow.

crates/auth/src/repository.rs 
use async_trait::async_trait; // add `async-trait = "0.1"` to Cargo.toml
          use sqlx::PgPool;
          use t1diaries_error::AppResult;
          use uuid::Uuid;

          use crate::model::UserRecord;

          #[async_trait]
          pub trait AuthRepository: Send + Sync {
          async fn find_by_email(&self, email: &str) -> AppResult<Option<UserRecord>>;
          async fn insert_user(&self, email: &str, password_hash: &str, display_name: &str) ->
          AppResult<UserRecord>;
          async fn mark_email_verified(&self, user_id: Uuid) -> AppResult<()>;
          async fn update_password_hash(&self, user_id: Uuid, new_hash: &str) -> AppResult<()>;
          }

          pub struct PgAuthRepository {
          pub pool: PgPool,
          }

          #[async_trait]
          impl AuthRepository for PgAuthRepository {
          async fn find_by_email(&self, email: &str) -> AppResult<Option<UserRecord>> {
          let user = sqlx::query_as!(
          UserRecord,
          r#"SELECT id, email, password_hash, display_name, role,
          email_verified_at, created_at
          FROM users WHERE email = $1"#,
          email
          )
          .fetch_optional(&self.pool)
          .await?;
          Ok(user)
          }

          async fn insert_user(&self, email: &str, password_hash: &str, display_name: &str) ->
          AppResult<UserRecord> {
          let user = sqlx::query_as!(
          UserRecord,
          r#"INSERT INTO users (email, password_hash, display_name)
          VALUES ($1, $2, $3)
          RETURNING id, email, password_hash, display_name, role,
          email_verified_at, created_at"#,
          email, password_hash, display_name
          )
          .fetch_one(&self.pool)
          .await?;
          Ok(user)
          }

          async fn mark_email_verified(&self, user_id: Uuid) -> AppResult<()> {
          sqlx::query!("UPDATE users SET email_verified_at = now() WHERE id = $1", user_id)
          .execute(&self.pool)
          .await?;
          Ok(())
          }

          async fn update_password_hash(&self, user_id: Uuid, new_hash: &str) -> AppResult<()> {
          sqlx::query!("UPDATE users SET password_hash = $1 WHERE id = $2", new_hash, user_id)
          .execute(&self.pool)
          .await?;
          Ok(())
          }
          }

7.2 Password hashing (argon2)

crates/auth/src/password.rs 
use argon2::{
          password_hash::{rand_core::OsRng, PasswordHash, PasswordHasher, PasswordVerifier, SaltString},
          Argon2,
          };
          use t1diaries_error::{AppError, AppResult};

          pub fn hash_password(plain: &str) -> AppResult<String> {
          let salt = SaltString::generate(&mut OsRng);
          let hash = Argon2::default()
          .hash_password(plain.as_bytes(), &salt)
          .map_err(|e| AppError::Internal(anyhow::anyhow!(e.to_string())))?;
          Ok(hash.to_string())
          }

          pub fn verify_password(plain: &str, hash: &str) -> AppResult<bool> {
          let parsed = PasswordHash::new(hash)
          .map_err(|e| AppError::Internal(anyhow::anyhow!(e.to_string())))?;
          Ok(Argon2::default().verify_password(plain.as_bytes(), &parsed).is_ok())
          }
Why argon2 over bcrypt? Argon2id (the default variant) is memory-hard, which makes GPU/ASIC brute-forcing significantly more expensive than bcrypt for equivalent CPU cost — it's the current OWASP recommendation for new applications.

7.3 JWT issuing & verification

crates/auth/src/jwt.rs 
use chrono::{Duration, Utc};
          use jsonwebtoken::{decode, encode, DecodingKey, EncodingKey, Header, Validation};
          use serde::{Deserialize, Serialize};
          use t1diaries_error::{AppError, AppResult};
          use uuid::Uuid;

          #[derive(Debug, Serialize, Deserialize)]
          pub struct Claims {
          pub sub: Uuid, // user id
          pub role: String,
          pub exp: i64, // expiry (unix timestamp)
          }

          pub fn issue_access_token(user_id: Uuid, role: &str, secret: &str, ttl_secs: i64) ->
          AppResult<String> {
          let claims = Claims {
          sub: user_id,
          role: role.to_string(),
          exp: (Utc::now() + Duration::seconds(ttl_secs)).timestamp(),
          };
          encode(&Header::default(), &claims, &EncodingKey::from_secret(secret.as_bytes()))
          .map_err(|e| AppError::Internal(anyhow::anyhow!(e)))
          }

          pub fn verify_access_token(token: &str, secret: &str) -> AppResult<Claims> {
          let data = decode::<Claims>(
          token,
          &DecodingKey::from_secret(secret.as_bytes()),
          &Validation::default(),
          )
          .map_err(|_| AppError::Unauthenticated)?;
          Ok(data.claims)
          }

The refresh token is a random opaque string (not a JWT) stored server-side in Redis as refresh:<token> -> user_id with a TTL, and set as an httpOnly, Secure, SameSite=Lax cookie via tower-cookies. This lets sign-out and password-reset instantly revoke sessions server-side — something a stateless JWT alone can't do.

7.4 Service layer (business logic — this is where "the work" happens)

crates/auth/src/service.rs 
use std::sync::Arc;
          use uuid::Uuid;
          use validator::Validate;
          use t1diaries_error::{AppError, AppResult};
          use t1diaries_email::EmailSender;

          use crate::{jwt, password, repository::AuthRepository};

          #[derive(Debug, Validate)]
          pub struct SignUpInput {
          #[validate(email)]
          pub email: String,
          #[validate(length(min = 8, message = "password must be at least 8 characters"))]
          pub password: String,
          #[validate(length(min = 2, max = 60))]
          pub display_name: String,
          }

          pub struct AuthTokens {
          pub access_token: String,
          pub refresh_token: String,
          }

          pub struct AuthService {
          pub repo: Arc<dyn AuthRepository>,
          pub email_sender: Arc<dyn EmailSender>,
          pub jwt_secret: String,
          pub access_ttl_secs: i64,
          }

          impl AuthService {
          pub async fn sign_up(&self, input: SignUpInput) -> AppResult<AuthTokens> {
          input.validate().map_err(|e| AppError::Validation(e.to_string()))?;

          if self.repo.find_by_email(&input.email).await?.is_some() {
          return Err(AppError::Conflict("email already registered".into()));
          }

          let hash = password::hash_password(&input.password)?;
          let user = self.repo.insert_user(&input.email, &hash, &input.display_name).await?;

          // Fire-and-forget style is fine here, but we `.await` for simplicity/reliability
          let verify_link = format!("https://t1diaries.app/verify?user={}", user.id);
          self.email_sender.send_verification_email(&user.email, &verify_link).await?;

          self.issue_tokens(user.id, &user.role)
          }

          pub async fn sign_in(&self, email: &str, password_plain: &str) -> AppResult<AuthTokens> {
          let user = self.repo.find_by_email(email).await?.ok_or(AppError::Unauthenticated)?;

          if !password::verify_password(password_plain, &user.password_hash)? {
          return Err(AppError::Unauthenticated);
          }

          self.issue_tokens(user.id, &user.role)
          }

          fn issue_tokens(&self, user_id: Uuid, role: &str) -> AppResult<AuthTokens> {
          let access_token = jwt::issue_access_token(user_id, role, &self.jwt_secret, self.access_ttl_secs)?;
          // In the real implementation, generate a random refresh token,
          // store it in Redis via crates/cache, and return it here.
          let refresh_token = Uuid::new_v4().to_string();
          Ok(AuthTokens { access_token, refresh_token })
          }
          }
Notice: nothing in service.rs knows it's being called from GraphQL. You can write #[tokio::test] async fn sign_up_rejects_duplicate_email() directly against AuthService with a mock AuthRepository, with zero HTTP or schema involved.

7.5 The GraphQL layer for auth (thin resolvers)

crates/auth/src/graphql/mutation.rs 
use async_graphql::{Context, InputObject, Object, Result, SimpleObject};

          use crate::service::{AuthService, SignUpInput as ServiceSignUpInput};

          #[derive(InputObject)]
          pub struct SignUpInput {
          pub email: String,
          pub password: String,
          pub display_name: String,
          }

          #[derive(InputObject)]
          pub struct SignInInput {
          pub email: String,
          pub password: String,
          }

          #[derive(SimpleObject)]
          pub struct AuthPayload {
          pub access_token: String,
          }

          #[derive(Default)]
          pub struct AuthMutation;

          #[Object]
          impl AuthMutation {
          async fn sign_up(&self, ctx: &Context<'_>, input: SignUpInput) -> Result<AuthPayload> {
          let service = ctx.data::<AuthService>()?;
          let tokens = service
          .sign_up(ServiceSignUpInput {
          email: input.email,
          password: input.password,
          display_name: input.display_name,
          })
          .await?;
          // NOTE: in apps/api, an axum middleware/extension sets the refresh
          // token as an httpOnly cookie on the outgoing response; the resolver
          // only returns the short-lived access token in the payload body.
          Ok(AuthPayload { access_token: tokens.access_token })
          }

          async fn sign_in(&self, ctx: &Context<'_>, input: SignInInput) -> Result<AuthPayload> {
          let service = ctx.data::<AuthService>()?;
          let tokens = service.sign_in(&input.email, &input.password).await?;
          Ok(AuthPayload { access_token: tokens.access_token })
          }

          async fn sign_out(&self, _ctx: &Context<'_>) -> Result<bool> {
          // revoke refresh token from Redis (via cache crate) — omitted for brevity
          Ok(true)
          }
          }

          pub use query::AuthQuery;

This query.rs / mutation.rs split, re-exported through a graphql/mod.rs, is the modularization pattern repeated identically in crates/users, crates/blogs, crates/stats, and crates/devices — see Step 8.

8. crates/users, blogs, stats, devices — the repeatable domain pattern

Read first: Rust Book Ch.13 — Functional Language Features (Iterators & Closures) — used heavily when mapping DB rows to GraphQL types and filtering/paginating collections; Ch.18 — Object-Oriented Programming Features for the trait-object (Arc<dyn Trait>) pattern used for every repository/service.

Every remaining domain crate follows the exact same four-file skeleton as crates/auth:

crates/<domain>/src/ ├── model.rs # sqlx::FromRow structs mapping DB rows ├── repository.rs # trait + Postgres impl (raw data access) ├── service.rs # business logic, calls repository, returns AppResult<T> ├── graphql/ │ ├── mod.rs # pub use query::*; pub use mutation::*; │ ├── query.rs # #[Object] impl exposing read fields │ └── mutation.rs # #[Object] impl exposing write fields (if any) └── lib.rs # pub mod model; pub mod repository; pub mod service; pub mod graphql;

8.1 Worked example: crates/devices (read-mostly domain)

crates/devices/src/model.rs 
use chrono::NaiveDate;
          use serde_json::Value as Json;

crates/devices/src/repository.rs 
use async_trait::async_trait;
          use sqlx::PgPool;
          use t1diaries_error::AppResult;

          use crate::model::DeviceRecord;

          #[async_trait]
          pub trait DeviceRepository: Send + Sync {
          async fn latest(&self, limit: i64) -> AppResult<Vec<DeviceRecord>>;
          async fn find_by_id(&self, id: uuid::Uuid) -> AppResult<Option<DeviceRecord>>;
          }

          pub struct PgDeviceRepository {
          pub pool: PgPool,
          }

          #[async_trait]
          impl DeviceRepository for PgDeviceRepository {
          async fn latest(&self, limit: i64) -> AppResult<Vec<DeviceRecord>> {
          let rows = sqlx::query_as!(
          DeviceRecord,
          r#"SELECT id, name, manufacturer, category, release_date,
          specs, image_url, description
          FROM devices ORDER BY release_date DESC LIMIT $1"#,
          limit
          )
          .fetch_all(&self.pool)
          .await?;
          Ok(rows)
          }

          async fn find_by_id(&self, id: uuid::Uuid) -> AppResult<Option<DeviceRecord>> {
          let row = sqlx::query_as!(
          DeviceRecord,
          r#"SELECT id, name, manufacturer, category, release_date,
          specs, image_url, description
          FROM devices WHERE id = $1"#,
          id
          )
          .fetch_optional(&self.pool)
          .await?;
          Ok(row)
          }
          }
crates/devices/src/graphql/query.rs 
use async_graphql::{Context, Object, Result, SimpleObject};
          use std::sync::Arc;

          use crate::repository::DeviceRepository;

          #[derive(SimpleObject)]
          pub struct Device {
          pub id: async_graphql::ID,
          pub name: String,
          pub manufacturer: String,
          pub category: String,
          pub release_date: String,
          }

          #[derive(Default)]
          pub struct DeviceQuery;

          #[Object]
          impl DeviceQuery {
          async fn latest_devices(&self, ctx: &Context<'_>, limit: Option<i64>) ->
          Result<Vec<Device>> {
          let repo = ctx.data::<Arc<dyn DeviceRepository>>()?;
          let rows = repo.latest(limit.unwrap_or(10)).await?;
          Ok(rows
          .into_iter()
          .map(|r| Device {
          id: r.id.into(),
          name: r.name,
          manufacturer: r.manufacturer,
          category: r.category,
          release_date: r.release_date.to_string(),
          })
          .collect())
          }
          }

Since devices has no user-facing mutation (curation is an internal/admin tool), its graphql/mod.rs only exports DeviceQuery — the root RootMutation in Step 9 simply doesn't include a DeviceMutation member. blogs and users follow the identical repository→service→graphql pattern but export both Query and Mutation, and stats is query-only (populated by a background job, not user mutations) — apply the same four-file skeleton to each using the field lists from the Roadmap document §6.

crates/graphql/src/schema.rs 
use async_graphql::{EmptySubscription, MergedObject, Schema};
          use std::sync::Arc;

          use t1diaries_auth::{graphql::{AuthMutation, AuthQuery}, service::AuthService};
          use t1diaries_blogs::graphql::{BlogMutation, BlogQuery};
          use t1diaries_devices::{graphql::DeviceQuery, repository::DeviceRepository};
          use t1diaries_stats::graphql::StatsQuery;
          use t1diaries_users::graphql::{UserMutation, UserQuery};

          #[derive(MergedObject, Default)]
          pub struct RootQuery(AuthQuery, UserQuery, BlogQuery, StatsQuery, DeviceQuery);

          #[derive(MergedObject, Default)]
          pub struct RootMutation(AuthMutation, UserMutation, BlogMutation);

          pub type AppSchema = Schema<RootQuery, RootMutation, EmptySubscription>;

          /// Everything a resolver anywhere in the tree might need to pull out of
          /// `ctx.data::<T>()`. Built once in apps/api::main and injected via `.data(...)`.
          pub struct AppState {
          pub auth_service: AuthService,
          pub device_repo: Arc<dyn DeviceRepository>,
          // ...user_service, blog_service, stats_service, etc.
          }

          pub fn build_schema(state: AppState) -> AppSchema {
          Schema::build(RootQuery::default(), RootMutation::default(), EmptySubscription)
          .data(state.auth_service)
          .data(state.device_repo)
          .finish()
          }
This file is intentionally the only place in the backend that knows about every domain crate at once. Everything else only knows about t1diaries-error and its own dependencies — this is what keeps cargo build -p t1diaries-blogs fast and decoupled.

use std::sync::Arc;

          use async_graphql::http::GraphiQLSource;
          use async_graphql_axum::GraphQL;
          use axum::{response::{Html, IntoResponse}, routing::get, Router};
          use tower_http::{cors::CorsLayer, trace::TraceLayer};

          use t1diaries_auth::service::AuthService;
          use t1diaries_config::AppConfig;
          use t1diaries_devices::repository::PgDeviceRepository;
          use t1diaries_graphql::{build_schema, AppState};

          #[tokio::main]
          async fn main() -> anyhow::Result<()> {
          tracing_subscriber::fmt::init();

          let config = AppConfig::load()?;
          let pool = t1diaries_db::create_pool(&config.database_url).await?;
          t1diaries_db::run_migrations(&pool).await?;
          let _redis = t1diaries_cache::create_pool(&config.redis_url)?;

          // Wire concrete implementations into each service (dependency injection,
          // done by hand — no DI framework needed at this scale).
          let auth_service = AuthService {
          repo: Arc::new(t1diaries_auth::repository::PgAuthRepository { pool: pool.clone() }),
          email_sender: Arc::new(t1diaries_email::SmtpEmailSender::new(&config)?),
          jwt_secret: config.jwt_secret.clone(),
          access_ttl_secs: config.access_token_ttl_secs,
          };
          let device_repo: Arc<dyn t1diaries_devices::repository::DeviceRepository> =
          Arc::new(PgDeviceRepository { pool: pool.clone() });

          let schema = build_schema(AppState { auth_service, device_repo });

          let app = Router::new()
          .route("/graphql", get(graphiql).post_service(GraphQL::new(schema)))
          .route("/healthz", get(|| async { "ok" }))
          .layer(CorsLayer::permissive()) // tighten to explicit origin in production
          .layer(TraceLayer::new_for_http());

          let listener = tokio::net::TcpListener::bind(("0.0.0.0", config.app_port)).await?;
          tracing::info!("listening on {}", listener.local_addr()?);
          axum::serve(listener, app).await?;
          Ok(())
          }

          async fn graphiql() -> impl IntoResponse {
          Html(GraphiQLSource::build().endpoint("/graphql").finish())
          }

import { useMutation, useQueryClient } from "@tanstack/react-query";
              import { gqlClient, setAccessToken } from "@t1diaries/graphql-client";
              import { SignInDocument } from "@t1diaries/graphql-client/generated"; // from codegen, see Step 13

              export function useSignIn() {
              const qc = useQueryClient();
              return useMutation({
              mutationFn: (input: { email: string; password: string }) =>
              gqlClient.request(SignInDocument, { input }),
              onSuccess: (data) => {
              setAccessToken(data.signIn.accessToken);
              qc.invalidateQueries({ queryKey: ["me"] });
              },
              });
              }
